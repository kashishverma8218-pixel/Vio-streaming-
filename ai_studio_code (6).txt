import React, { useState } from 'react';
import { VioLogo } from './components/VioLogo';
import { VideoPlayer } from './components/VideoPlayer';
import { VideoCard } from './components/VideoCard';
import { UploadManager } from './components/UploadManager';
import { CHANNELS, INITIAL_VIDEOS, INITIAL_COMMENTS } from './data/mockData';
import { VideoItem, Channel } from './types/video';
import { Search, Mic, Moon, Sun, Plus, Home, Flame, Bell, Settings, ThumbsUp } from 'lucide-react';

export default function App() {
  const [isDarkMode, setIsDarkMode] = useState<boolean>(true);
  const [videos, setVideos] = useState<VideoItem[]>(INITIAL_VIDEOS);
  const [channels] = useState<Record<string, Channel>>(CHANNELS);
  const [activeVideo, setActiveVideo] = useState<VideoItem>(INITIAL_VIDEOS[0]);
  const [searchQuery, setSearchQuery] = useState('');
  const [selectedCategory, setSelectedCategory] = useState('All');
  const [isUploadOpen, setIsUploadOpen] = useState(false);

  const categories = ['All', 'Animation', 'Sci-Fi', 'VFX', '3D Blender', 'Shorts', 'Live'];

  const filteredVideos = videos.filter((v) => {
    const matchesCategory = selectedCategory === 'All' || v.category === selectedCategory;
    const matchesSearch = v.title.toLowerCase().includes(searchQuery.toLowerCase());
    return matchesCategory && matchesSearch;
  });

  const handlePublishFromManager = (newVidData: Partial<VideoItem>) => {
    const newVid: VideoItem = {
      id: `custom-${Date.now()}`,
      title: newVidData.title || 'Untitled Stream',
      description: 'Streamed on Vio streaming platform.',
      url: 'https://commondatastorage.googleapis.com/gtv-videos-bucket/sample/TearsOfSteel.mp4',
      poster: 'https://images.unsplash.com/photo-1579783900882-c0d3dad7b119?auto=format&fit=crop&w=800&q=80',
      channelId: 'channel-blender',
      views: 1,
      likes: 1,
      dislikes: 0,
      publishedAt: 'Just now',
      duration: 320,
      category: 'Animation',
      tags: ['VioLive'],
      visibility: newVidData.visibility || 'public',
    };
    setVideos([newVid, ...videos]);
    setActiveVideo(newVid);
    setIsUploadOpen(false);
  };

  return (
    <div className={`min-h-screen ${isDarkMode ? 'dark bg-slate-950 text-white' : 'bg-slate-50 text-slate-900'} flex flex-col`}>
      {/* HEADER / NAVBAR */}
      <header className="sticky top-0 z-40 flex items-center justify-between px-4 py-2.5 bg-white/90 dark:bg-slate-950/90 backdrop-blur-md border-b border-slate-200 dark:border-slate-800">
        <div className="flex items-center gap-3">
          <VioLogo size="md" />
        </div>

        {/* Center Search Bar */}
        <div className="hidden sm:flex flex-1 max-w-lg mx-6 items-center">
          <div className="relative w-full flex items-center">
            <input
              type="text"
              placeholder="Search videos, reels, channels on Vio streaming..."
              value={searchQuery}
              onChange={(e) => setSearchQuery(e.target.value)}
              className="w-full px-4 py-2 rounded-l-full border border-r-0 border-slate-300 dark:border-slate-700 bg-slate-50 dark:bg-slate-900 text-xs focus:outline-none"
            />
            <button className="px-4 py-2 rounded-r-full bg-slate-100 dark:bg-slate-800 border border-slate-300 dark:border-slate-700 text-slate-500 hover:text-red-500">
              <Search className="h-4 w-4" />
            </button>
          </div>
        </div>

        {/* Right Action Icons */}
        <div className="flex items-center gap-2">
          <button
            onClick={() => setIsUploadOpen(true)}
            className="flex items-center gap-1.5 px-3 py-1.5 rounded-full bg-red-600 text-white text-xs font-bold hover:bg-red-500 shadow-sm"
          >
            <Plus className="h-3.5 w-3.5" />
            <span className="hidden sm:inline">Upload</span>
          </button>
          <button
            onClick={() => setIsDarkMode(!isDarkMode)}
            className="p-2 rounded-full hover:bg-slate-100 dark:hover:bg-slate-800 text-slate-600 dark:text-slate-300"
          >
            {isDarkMode ? <Sun className="h-4 w-4 text-amber-400" /> : <Moon className="h-4 w-4" />}
          </button>
        </div>
      </header>

      {/* CATEGORY BAR */}
      <div className="sticky top-[57px] z-30 px-4 py-2 bg-white/80 dark:bg-slate-950/80 backdrop-blur-xs border-b border-slate-200 dark:border-slate-800 flex gap-2 overflow-x-auto no-scrollbar">
        {categories.map((cat) => (
          <button
            key={cat}
            onClick={() => setSelectedCategory(cat)}
            className={`px-3.5 py-1 rounded-full text-xs font-semibold whitespace-nowrap transition-colors ${
              selectedCategory === cat
                ? 'bg-slate-900 text-white dark:bg-white dark:text-slate-900'
                : 'bg-slate-100 dark:bg-slate-900 text-slate-600 dark:text-slate-400 hover:bg-slate-200 dark:hover:bg-slate-800'
            }`}
          >
            {cat}
          </button>
        ))}
      </div>

      {/* MAIN CONTENT AREA */}
      <main className="flex-1 max-w-7xl w-full mx-auto p-4 sm:p-6 space-y-8">
        {/* ACTIVE CINEMA PLAYER */}
        <section className="space-y-4">
          <VideoPlayer video={activeVideo} />
          <div className="space-y-1">
            <h1 className="text-xl sm:text-2xl font-black">{activeVideo.title}</h1>
            <div className="flex items-center justify-between text-xs text-slate-400 pb-3 border-b border-slate-200 dark:border-slate-800">
              <span>{channels[activeVideo.channelId]?.name} · {activeVideo.views.toLocaleString()} views</span>
              <button className="flex items-center gap-1.5 px-3 py-1.5 rounded-full bg-slate-100 dark:bg-slate-800 hover:bg-slate-200 text-slate-800 dark:text-slate-200 font-bold">
                <ThumbsUp className="h-3.5 w-3.5" />
                <span>{activeVideo.likes.toLocaleString()}</span>
              </button>
            </div>
          </div>
        </section>

        {/* FEED GRID */}
        <section className="space-y-4">
          <h2 className="text-lg font-extrabold">Trending on Vio streaming</h2>
          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-6">
            {filteredVideos.map((video) => (
              <VideoCard
                key={video.id}
                video={video}
                channel={channels[video.channelId]}
                onPlay={(v) => {
                  setActiveVideo(v);
                  window.scrollTo({ top: 0, behavior: 'smooth' });
                }}
              />
            ))}
          </div>
        </section>
      </main>

      {/* UPLOAD MODAL (Featuring the requested useReducer UploadManager) */}
      {isUploadOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/80 backdrop-blur-sm animate-in fade-in">
          <UploadManager
            onPublish={handlePublishFromManager}
            onCancel={() => setIsUploadOpen(false)}
          />
        </div>
      )}
    </div>
  );
}
