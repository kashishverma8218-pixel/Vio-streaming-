import React from 'react';
import { Channel, VideoItem } from '../types/video';

interface VideoCardProps {
  video: VideoItem;
  channel?: Channel;
  onPlay: (video: VideoItem) => void;
}

export const VideoCard: React.FC<VideoCardProps> = ({ video, channel, onPlay }) => {
  const formatTime = (seconds: number) => {
    const mins = Math.floor(seconds / 60);
    const secs = Math.floor(seconds % 60);
    return `${mins}:${secs.toString().padStart(2, '0')}`;
  };

  return (
    <div
      onClick={() => onPlay(video)}
      className="group flex flex-col gap-3 cursor-pointer select-none"
    >
      {/* Thumbnail Container */}
      <div className="relative aspect-video w-full overflow-hidden rounded-2xl bg-slate-100 dark:bg-slate-900 shadow-sm transition-transform duration-200 group-hover:scale-[1.02]">
        <img
          src={video.poster}
          alt={video.title}
          className="h-full w-full object-cover"
          loading="lazy"
        />
        {/* Contrast Scrim */}
        <div className="absolute inset-0 bg-gradient-to-t from-black/85 via-black/20 to-transparent pointer-events-none" />
        
        {/* WCAG High-Contrast Duration Badge */}
        <span className="absolute bottom-2 right-2 px-1.5 py-0.5 rounded-md bg-black/90 text-white font-mono text-[11px] font-semibold ring-1 ring-white/20">
          {formatTime(video.duration)}
        </span>
      </div>

      {/* Metadata */}
      <div className="flex gap-3 px-0.5">
        <img
          src={channel?.avatar || 'https://images.unsplash.com/photo-1534528741775-53994a69daeb?auto=format&fit=crop&w=120&q=80'}
          alt={channel?.name}
          className="h-9 w-9 rounded-full object-cover ring-1 ring-slate-200 dark:ring-slate-800 shrink-0"
        />
        <div className="flex flex-col min-w-0">
          <h3 className="text-sm font-bold text-slate-900 dark:text-white line-clamp-2 leading-snug group-hover:text-red-500 transition-colors">
            {video.title}
          </h3>
          <span className="text-xs text-slate-500 dark:text-slate-400 mt-1 truncate">
            {channel?.name || 'Vio Creator'}
          </span>
          <div className="flex items-center gap-1.5 text-[11px] text-slate-400">
            <span>{(video.views / 1000).toFixed(1)}k views</span>
            <span>•</span>
            <span>{video.publishedAt}</span>
          </div>
        </div>
      </div>
    </div>
  );
};
